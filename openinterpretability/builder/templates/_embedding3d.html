0.3 : 0.25, dark ? 0x000000 : 0x444444); spaceGroup.add(cloudPoints, cloudShadows); setVisible(spaceGroup, view === "space"); } // The sentence: a sphere per token, its shadow, its label, the path, and its trail. let sentence = [], comparison = []; function tokensOf(p, rings) { const n = p.tokens.length, out = []; const path = new THREE.Line(new THREE.BufferGeometry(), rings ? "Pause" : "Play the layers"; $("route3d-compare").addEventListener("change", (e) => { compare = +e.target.value; show(+$("route3d-prompt").value); }); if (hasSpace) loadCloud().catch((error) => { console.error("route3d: no embedding space", error); });
